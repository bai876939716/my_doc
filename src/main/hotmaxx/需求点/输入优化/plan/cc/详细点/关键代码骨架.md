## 一、状态机 / MCP / 类别识别

```python
# 1. 状态机引擎（转移表配置化 · 11 状态）
TRANSITIONS = {
    "INIT":      {"text": "A_WAIT", "card": "B_LOCK"},
    "A_WAIT":    {"ok": "A_ENUM", "fail": "FALLBACK"},   # ok=识别出候选；fail=失败/候选全被白名单过滤
    "A_ENUM":    {"pick": "DETAIL"},
    "B_LOCK":    {"auto": "DETAIL"},
    "DETAIL":    {"ok": "PHOTO", "missing": "DETAIL"},
    "PHOTO":     {"pass": "CONTACT", "lack": "PHOTO"},
    "CONTACT":   {"confirm": "CONFIRM"},
    "CONFIRM":   {"submit": "DONE", "recat": "PHOTO"},
    "DONE":      {},
    "FALLBACK":  {"pick": "DETAIL"},
    "EXITED":    {},   # 主动退出终态
}
EXIT_EVENTS = {"exit_keep": "EXITED", "exit_abandon": "EXITED"}  # DONE 之外的任意状态均可触发

def step(state, event, ctx):
    nxt = EXIT_EVENTS.get(event) or TRANSITIONS[state].get(event)
    if nxt: run_state_handler(nxt, ctx); ctx.state = nxt; draft_save(ctx)
    return nxt

# 2. MCP Server（fastmcp · Streamable HTTP · 独立进程）
@mcp.tool()
def get_category_dict(store_no: str = "") -> dict:
    """返回类别树、Top6 高频卡片文案、各类别拍摄清单"""
    return dict_repo.get(store_no)

if __name__ == "__main__":
    mcp.run(transport="streamable-http")   # 单端点，默认 http://0.0.0.0:8000/mcp

# 2b. Agent 侧 MCP 客户端（官方 SDK · streamable http client；长会话可复用 MCP-Session-Id）
from mcp import ClientSession
from mcp.client.streamable_http import streamablehttp_client

async def call_tool(name, args: dict):
    async with streamablehttp_client(MCP_URL) as (read, write, _):   # MCP_URL = http://mcp-svc:8000/mcp
        async with ClientSession(read, write) as sess:
            await sess.initialize()
            return await sess.call_tool(name, args)

# 3. 类别识别 + 防幻觉校验（无置信度：识别成功一律 Top3 卡，失败才 FALLBACK）
def recognize(text, dict_list):
    out = llm_json(CAT_PROMPT, user_text=text, category_dict=dict_list)    # LLM-XXX
    valid = [c for c in out["candidates"] if c["name"] in set(dict_list)]  # 枚举白名单过滤
    if not valid: return fallback_full_list(dict_list)
    return valid[:3]
```
